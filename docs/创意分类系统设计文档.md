**版本**: v1.0

## 📋 概述

创意空间采用**标签驱动的灵活分类系统**，通过双维度分类体系实现对创意内容的精准分类和高效检索。

### 设计目标
- **灵活性**: 支持复杂的多维分类需求
- **扩展性**: 易于添加新的分类和标签
- **用户友好**: 直观的导航和筛选体验
- **性能优化**: 高效的查询和筛选算法

---

## 🎯 核心设计理念

### 双维度分类体系

#### 主分类维度 (Category)
按创意成熟度划分，体现创意的发展阶段：

```
idea      → 创意想法 (概念阶段)
prototype → 原型Demo (可演示阶段)  
project   → 完整项目 (成品阶段)
resource  → 创意资源 (素材阶段)
```

#### 标签维度 (Tags)
多维属性标记，支持任意组合：

```
领域标签: game, app, creative, hardware, art, design
技术标签: ai, blockchain, vr, ar, mobile, web, unity
状态标签: concept, development, testing, released, maintained
类型标签: 2d, 3d, animation, sound, ui, character, scene
```

### 分类映射关系

```
游戏原型 = category:'prototype' + tags:['game']
游戏资源 = category:'resource' + tags:['game', 'character'] 
AI创意项目 = category:'project' + tags:['creative', 'ai']
区块链应用 = category:'project' + tags:['app', 'blockchain']
```

#### 组合筛选逻辑
```typescript
// 示例：筛选"游戏领域的AI技术原型"
const filters = {
  maturity: ['prototype'],    // 成熟度：原型
  domain: ['game'],          // 领域：游戏
  tech: ['ai']              // 技术：AI
};

// 筛选实现
function filterCreatives(creatives: Creative[], filters: FilterOptions): Creative[] {
  return creatives.filter(creative => {
    // 成熟度筛选
    if (filters.maturity?.length > 0) {
      if (!filters.maturity.includes(creative.category)) return false;
    }
    
    // 领域筛选 (OR逻辑)
    if (filters.domain?.length > 0) {
      const hasMatchingDomain = filters.domain.some(domain => 
        creative.tags.includes(domain)
      );
      if (!hasMatchingDomain) return false;
    }
    
    // 技术筛选 (OR逻辑)
    if (filters.tech?.length > 0) {
      const hasMatchingTech = filters.tech.some(tech => 
        creative.tags.includes(tech)
      );
      if (!hasMatchingTech) return false;
    }
    
    // 状态筛选 (OR逻辑)
    if (filters.status?.length > 0) {
      const hasMatchingStatus = filters.status.some(status => 
        creative.tags.includes(status)
      );
      if (!hasMatchingStatus) return false;
    }
    
    return true;
  });
}
```

### 3. 搜索功能

```typescript
function searchCreatives(creatives: Creative[], searchTerm: string): Creative[] {
  if (!searchTerm.trim()) return creatives;
  
  const term = searchTerm.toLowerCase();
  
  return creatives.filter(creative => {
    // 标题匹配
    if (creative.title.toLowerCase().includes(term)) return true;
    
    // 描述匹配
    if (creative.description.toLowerCase().includes(term)) return true;
    
    // 标签匹配
    if (creative.tags.some(tag => tag.toLowerCase().includes(term))) return true;
    
    return false;
  });
}
```

```move
// 主分类常量
const CATEGORY_IDEA: vector<u8> = b"idea";
const CATEGORY_PROTOTYPE: vector<u8> = b"prototype";
const CATEGORY_PROJECT: vector<u8> = b"project";
const CATEGORY_RESOURCE: vector<u8> = b"resource";
```

### 查询接口

```move
// 按分类匹配
public fun is_category_match(creative: &Creative, category: &String): bool {
    &creative.category == category
}

// 按标签匹配 (支持AND/OR逻辑)
public fun has_tags(creative: &Creative, tags: &vector<String>, match_all: bool): bool {
    let creative_tags = &creative.tags;
    let mut matched_count = 0;
    let target_count = vector::length(tags);
    
    let mut i = 0;
    while (i < target_count) {
        let target_tag = vector::borrow(tags, i);
        if (vector::contains(creative_tags, target_tag)) {
            matched_count = matched_count + 1;
            if (!match_all) {
                return true // OR查询，匹配一个即返回
            }
        };
        i = i + 1;
    };
    
    if (match_all) {
        matched_count == target_count // AND查询，必须全部匹配
    } else {
        false // OR查询但没有匹配任何标签
    }
}

// 复合查询：分类 + 标签
public fun matches_category_and_tags(
    creative: &Creative, 
    category: &String, 
    tags: &vector<String>,
    match_all_tags: bool
): bool {
    is_category_match(creative, category) && has_tags(creative, tags, match_all_tags)
}

// 成熟度筛选：检查是否属于指定成熟度分类
public fun is_maturity_match(creative: &Creative, categories: &vector<String>): bool {
    let creative_category = &creative.category;
    vector::contains(categories, creative_category)
}

// 分类验证
public fun is_valid_category(category: &String): bool {
    let idea = utf8(CATEGORY_IDEA);
    let prototype = utf8(CATEGORY_PROTOTYPE);
    let project = utf8(CATEGORY_PROJECT);
    let resource = utf8(CATEGORY_RESOURCE);
    
    category == &idea || category == &prototype || category == &project || category == &resource
}
```

```typescript
// CategoryFilter.tsx
const CategoryFilter: React.FC = ({ onFilterChange, selectedFilters }) => {
  const [activeTab, setActiveTab] = useState('maturity');
  
  const handleFilterChange = (filterType: string, value: string, checked: boolean) => {
    const currentFilters = selectedFilters[filterType] || [];
    const newFilters = checked 
      ? [...currentFilters, value]
      : currentFilters.filter(item => item !== value);
    
    onFilterChange({
      ...selectedFilters,
      [filterType]: newFilters
    });
  };
  
  return (
    <div className="bg-white rounded-lg shadow-md p-6">
      {/* 标签页 */}
      <div className="flex space-x-1 mb-4">
        {Object.entries(FILTER_CONFIG).map(([key, config]) => (
          <button
            key={key}
            onClick={() => setActiveTab(key)}
            className={activeTab === key ? 'active' : ''}
          >
            {config.title}
          </button>
        ))}
      </div>
      
      {/* 筛选选项 */}
      <div>
        {FILTER_CONFIG[activeTab]?.options.map(option => (
          <label key={option.value}>
            <input
              type="checkbox"
              checked={selectedFilters[activeTab]?.includes(option.value)}
              onChange={(e) => handleFilterChange(activeTab, option.value, e.target.checked)}
            />
            {option.label}
          </label>
        ))}
      </div>
    </div>
  );
};
```
